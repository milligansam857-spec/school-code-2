from dataclasses import dataclass from datetime import datetime, timedelta from typing import List @dataclass class Task: name: str deadline: datetime priority_level: int # 1 (Highest) to 5 (Lowest) estimated_hours: float def calculate_urgency_score(self) -> float: """ Calculates a dynamic urgency score. Lower score = higher urgency for the reshuffle algorithm. """ days_remaining = (self.deadline - datetime.now()).total_seconds() / 86400 # Prevent division by zero and heavily penalize overdue tasks days_remaining = max(days_remaining, 0.1) # Algorithm balances the static priority level with the approaching deadline # and the time required to complete it. return (self.priority_level * days_remaining) / self.estimated_hours class ScheduleManager: def __init__(self): self.tasks: List[Task] = [] def add_task(self, task: Task): self.tasks.append(task) def trigger_reshuffle(self) -> List[Task]: """ Executes a one-click reshuffle, instantly re-sorting the active queue based on the most current urgency scores. """ # Sort tasks in-place based on the calculated score self.tasks.sort(key=lambda task: task.calculate_urgency_score()) return self.tasks # --- Example Usage --- if __name__ == "__main__": manager = ScheduleManager() # Adding sample tasks manager.add_task(Task("Wireframe UI", datetime.now() + timedelta(days=2), 2, 4.0)) manager.add_task(Task("Database Schema", datetime.now() + timedelta(days=1), 1, 6.0)) manager.add_task(Task("Write Readme", datetime.now() + timedelta(days=7), 4, 1.0)) print("Executing dynamic reshuffle...") optimized_queue = manager.trigger_reshuffle() for i, task in enumerate(optimized_queue, 1): print(f"{i}. {task.name} (Deadline: {task.deadline.strftime('%b %d')})")
